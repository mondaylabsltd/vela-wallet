const out=document.getElementById('out');
const S={results:[],events:[]};
const show=()=>out.textContent=JSON.stringify(S,null,1);

const P=window.ethereum;
if(P){for(const e of ['accountsChanged','chainChanged','connect','disconnect'])P.on(e,d=>{S.events.push({t:t(),event:e,data:d});show();});}
const FOUNDER='0x76875e38fc6Bc2dEDCaed807cE00782DB5C0D141';

async function ask(label,method,params){const r={t:t(),label,method};S.results.push(r);show();
 try{r.result=await P.request({method,params:params??[]});r.ok=true;}catch(e){r.ok=false;r.code=e.code;r.message=e.message;}
 r.done=t();show();}
const $=id=>document.getElementById(id);
